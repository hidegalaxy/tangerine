function configurar() {
  const props = PropertiesService.getScriptProperties();
  if (props.getProperty('HOJA')) {
    try {
      SpreadsheetApp.openById(props.getProperty('HOJA'));
    } catch (err) {
      ['HOJA', 'CLAVE', 'CAL', 'CALENDARIO'].forEach(p => props.deleteProperty(p));
    }
  }
  if (!props.getProperty('HOJA')) {
    const nuevo = SpreadsheetApp.create('Tangerine');
    nuevo.setSpreadsheetTimeZone(TZ);
    nuevo.setSpreadsheetLocale('es_ES');
    nuevo.getSheets()[0].setName('Registro');
    props.setProperty('HOJA', nuevo.getId());
  }
  const ss = libro_();
  Object.keys(ESQUEMA).forEach(nombre => {
    const h = ss.getSheetByName(nombre) || ss.insertSheet(nombre);
    const ultima = h.getLastColumn();
    const cab = ultima ? h.getRange(1, 1, 1, ultima).getValues()[0].map(String) : [];
    const faltan = ESQUEMA[nombre].filter(c => cab.indexOf(c) < 0);
    if (faltan.length) h.getRange(1, cab.length + 1, 1, faltan.length).setValues([faltan]);
    h.setFrozenRows(1);
    // Fechas y JSON como texto: que la hoja no los convierta en fechas ni números.
    h.getRange('A:E').setNumberFormat('@');
  });
  const cal = CalendarApp.getDefaultCalendar();
  if (!props.getProperty('CAL')) props.setProperty('CAL', cal.getId());
  if (!props.getProperty('CALENDARIO')) props.setProperty('CALENDARIO', 'si');
  if (!props.getProperty('CLAVE')) props.setProperty('CLAVE', Utilities.getUuid().replace(/-/g, '').slice(0, 16));
  // Cada madrugada pone en el calendario los días que entran en las dos semanas siguientes.
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'diario').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('diario').timeBased().everyDays(1).atHour(4).inTimezone(TZ).create();
  // Cada 15 minutos mira si el reloj ha subido algo nuevo a Garmin (si hay clave).
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'cadaRato').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('cadaRato').timeBased().everyMinutes(15).create();
  sincronizar_();
  Logger.log('Calendario: ' + cal.getName());
  Logger.log('Hoja: ' + ss.getUrl());
  Logger.log('Clave: ' + props.getProperty('CLAVE'));
}

function diario() {

}

function cadaRato() {
  garmin_(3);
}

// GET ?k=CLAVE                     → app
// GET ?k=CLAVE&accion=datos        → registro en JSON (para la web propia)
// GET ?k=CLAVE&accion=hoy          → qué toca hoy, en texto (atajo y Siri)
// GET ?k=CLAVE&accion=widget       → datos del widget de Scriptable
// GET ?k=CLAVE&accion=scriptable   → código del widget, listo para copiar
// GET (sin clave) o ?accion=instalar → página para instalar Tangerine en el iPhone
function doGet(e) {
  const p = e.parameter;
  if (p.accion === 'instalar' || (!p.k && !p.accion)) return paginaInstalar_(p.k);
  if (!claveValida_(p.k)) return p.accion === 'datos' || p.accion === 'widget' ? json_({ error: 'Clave incorrecta' }) : texto_('Clave incorrecta');
  if (p.accion === 'hoy') return texto_(textoHoy_());
  if (p.accion === 'widget') return json_(widget_(p.k));
  if (p.accion === 'scriptable') return texto_(SCRIPTABLE_.replace('__URL__', ScriptApp.getService().getUrl() + '?k=' + p.k + '&accion=widget'));
  if (p.accion === 'datos') {
    try {
      return json_(datos_());
    } catch (err) {
      return json_({ error: err.message });
    }
  }
  const t = HtmlService.createTemplateFromFile('Panel');
  t.clave = p.k;
  return t.evaluate()
    .setTitle('Tangerine')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover');
}

// POST { api: 'accion', k, que, d } desde la web propia → JSON.
function doPost(e) {
  let d;
  try {
    d = JSON.parse(e.postData.contents);
  } catch (err) {
    d = e.parameter || {};
  }
  if (!d.k && e.parameter && e.parameter.k) d.k = e.parameter.k;
  try {
    return json_(accion(d.k, d.que, d.d));
  } catch (err) {
    return json_({ error: err.message });
  }
}

// ---------- Llamadas de la app ----------

function datos(k) {
  comprobar_(k);
  return datos_();
}

function accion(k, que, d) {
  comprobar_(k);
  if (!ACCIONES_[que]) throw new Error('Acción desconocida: ' + que);
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  let r;
  try {
    r = ACCIONES_[que](d || {});
  } finally {
    lock.releaseLock();
  }
  return { msg: (r && r.msg) || '', ids: (r && r.ids) || [], datos: datos_() };
}

const TIPOS_REGISTRO_ = ['sesion', 'rodilla', 'mover', 'saltar', 'pausa', 'quitarPausa', 'deshacer', 'ignorar'];

const ACCIONES_ = {
  // Añade entradas del móvil. Las que ya estaban (mismo id) se ignoran: reenviar no duplica.
  anadir: d => {
    const entradas = Array.isArray(d.entradas) ? d.entradas : [];
    const ya = new Set(leer_('Registro').map(r => String(r.id)));
    const ids = [];
    let nuevas = 0;
    entradas.forEach(en => {
      const id = String(en.id || '');
      if (!/^[\w-]{4,40}$/.test(id)) return;
      if (!TIPOS_REGISTRO_.includes(en.tipo)) return;
      ids.push(id);
      if (ya.has(id)) return;
      const resto = {};
      Object.keys(en).forEach(c => { if (['id', 'cuando', 'tipo'].indexOf(c) < 0) resto[c] = en[c]; });
      const cuando = /^\d{4}-\d{2}-\d{2}T/.test(String(en.cuando)) ? String(en.cuando) : new Date().toISOString();
      anadir_('Registro', { id, cuando, tipo: en.tipo, dia: String(en.dia || en.de || en.desde || ''), datos: JSON.stringify(resto) });
      ya.add(id);
      nuevas++;
    });
    if (nuevas) sincronizar_();
    return { ids, msg: nuevas ? 'Guardado' : '' };
  },
  ajuste: d => {
    const props = PropertiesService.getScriptProperties();
    if (d.calendario != null) {
      props.setProperty('CALENDARIO', d.calendario ? 'si' : 'no');
      sincronizar_();
    }
    if (d.garmin != null) {
      CacheService.getScriptCache().remove('GARMIN_TOKEN');
      const clave = String(d.garmin).trim();
      if (!clave) {
        props.deleteProperty('GARMIN_KEY');
        props.deleteProperty('GARMIN_ESTADO');
        return { msg: 'Garmin desconectado' };
      }
      props.setProperty('GARMIN_KEY', clave);
      const r = garmin_(90);
      if (!r.ok) {
        props.deleteProperty('GARMIN_KEY');
        throw new Error(r.error);
      }
      return { msg: 'Garmin conectado: ' + r.nuevas + ' actividades' };
    }
    return { msg: 'Hecho' };
  },
  garmin: () => {
    const r = garmin_(14);
    if (!r.ok) throw new Error(r.error || 'Garmin no está conectado');
    return { msg: r.nuevas ? r.nuevas + ' actividades nuevas' : 'Nada nuevo en Garmin' };
  },
};

function datos_() {
  const props = PropertiesService.getScriptProperties();
  // Al abrir la app, si hace rato que no se miró Garmin, se mira ya.
  if (props.getProperty('GARMIN_KEY')) {
    let e = {};
    try { e = JSON.parse(props.getProperty('GARMIN_ESTADO') || '{}'); } catch (err) {}
    if (!e.cuando || Date.now() - new Date(e.cuando).getTime() > 10 * 60000) garmin_(3);
  }
  let estado = {};
  try { estado = JSON.parse(props.getProperty('GARMIN_ESTADO') || '{}'); } catch (err) {}
  return {
    garmin: Object.assign({ conectado: !!props.getProperty('GARMIN_KEY') }, estado),
    log: registro_(),
    hoy: diaDe_(new Date()),
    calendario: PropertiesService.getScriptProperties().getProperty('CALENDARIO') !== 'no',
  };
}

function registro_() {
  return leer_('Registro').map(r => {
    let resto = {};
    try { resto = JSON.parse(String(r.datos || '{}')); } catch (err) {}
    const cuando = r.cuando instanceof Date ? r.cuando.toISOString() : String(r.cuando);
    return Object.assign(resto, { id: String(r.id), cuando, tipo: String(r.tipo) });
  });
}

// ---------- Garmin ----------

// Trae las actividades de los últimos `dias` días que aún no están en el registro.
// Las de fuerza hechas con un entreno de Tangerine se apuntan solas como sesión, serie por serie.
function garmin_(dias) {
  const props = PropertiesService.getScriptProperties();
  const clave = props.getProperty('GARMIN_KEY');
  if (!clave) return { ok: false, error: 'Garmin no está conectado' };
  const guardarEstado = o => props.setProperty('GARMIN_ESTADO', JSON.stringify(Object.assign({ cuando: new Date().toISOString() }, o)));
  try {
    const token = tokenGarmin_(clave);
    const hoy = diaDe_(new Date());
    const r = pedirGarmin_(token, '/activitylist-service/activities/search/activities?start=0&limit=50&startDate=' + PLAN.diaMas(hoy, -dias) + '&endDate=' + hoy);
    const j = JSON.parse(r.getContentText());
    const lista = Array.isArray(j) ? j : (j.activityList || j.activities || []);
    const ya = new Set(leer_('Registro').map(x => String(x.id)));
    let nuevas = 0;
    // De la más antigua a la más nueva, para que las sesiones del reloj entren en orden.
    lista.slice().reverse().forEach(a => {
      const g = actividadGarmin_(a);
      if (!g || ya.has(g.id)) return;
      const sesion = g.actividad.tipo === 'fuerza' || g.actividad.tipo === 'flex' ? leerSesionGarmin_(token, g) : null;
      anadir_('Registro', { id: g.id, cuando: new Date().toISOString(), tipo: 'garmin', dia: g.dia, datos: JSON.stringify({ dia: g.dia, actividad: g.actividad }) });
      ya.add(g.id);
      if (sesion && !ya.has(sesion.id)) {
        const resto = Object.assign({}, sesion);
        delete resto.id;
        anadir_('Registro', { id: sesion.id, cuando: new Date().toISOString(), tipo: 'sesion', dia: sesion.dia, datos: JSON.stringify(resto) });
        ya.add(sesion.id);
      }
      nuevas++;
    });
    guardarEstado({ ok: true, ultima: lista.length ? String(lista[0].startTimeLocal || '').slice(0, 10) : '' });
    if (nuevas) sincronizar_();
    return { ok: true, nuevas };
  } catch (err) {
    guardarEstado({ ok: false, error: err.message });
    return { ok: false, error: err.message };
  }
}

function tokenGarmin_(clave) {
  const cache = CacheService.getScriptCache(), t = cache.get('GARMIN_TOKEN');
  if (t) return t;
  const r = UrlFetchApp.fetch(GARMIN_URL + '/personalapitoken-service/personal-api/personal-api-key/token', {
    method: 'post', headers: { Authorization: 'ApiKey ' + clave, Accept: '*/*' }, muteHttpExceptions: true,
  });
  const c = r.getResponseCode();
  if (c === 401 || c === 403) throw new Error('La clave de Garmin no vale o ha caducado');
  if (c !== 200) throw new Error('Garmin no da acceso ahora (' + c + ')');
  const token = r.getContentText().trim();
  cache.put('GARMIN_TOKEN', token, 6 * 3600 - 60); // el pase dura 24 h; la caché, como mucho 6
  return token;
}

function pedirGarmin_(token, ruta) {
  const r = UrlFetchApp.fetch(GARMIN_URL + ruta, { headers: { Authorization: 'Bearer ' + token, Accept: 'application/json' }, muteHttpExceptions: true });
  const c = r.getResponseCode();
  if (c === 401) {
    CacheService.getScriptCache().remove('GARMIN_TOKEN');
    throw new Error('Garmin ha caducado el pase; se pedirá otro');
  }
  if (c !== 200) throw new Error('Garmin ha contestado ' + c);
  return r;
}

// Resumen de una actividad de la lista de Garmin.
function actividadGarmin_(a) {
  const inicio = String(a.startTimeLocal || '');
  const dia = inicio.slice(0, 10);
  if (!a.activityId || !/^\d{4}-\d{2}-\d{2}$/.test(dia)) return null;
  const tk = String((a.activityType && a.activityType.typeKey) || '');
  const tipo = /running/.test(tk) ? 'run' : /strength/.test(tk) ? 'fuerza' : /yoga|pilates|flexib|mobility|breath/.test(tk) ? 'flex' : 'otro';
  const seg = Number(a.movingDuration || a.duration || a.elapsedDuration || 0), v = Number(a.averageSpeed || 0);
  const act = {
    id: String(a.activityId), nombre: String(a.activityName || ''), tipo, clave: tk, hora: inicio.slice(11, 16),
    min: Math.round(seg / 60), km: a.distance ? Math.round(a.distance / 10) / 100 : 0, ritmo: tipo === 'run' && v > 0 ? Math.round(1000 / v) : null,
    fc: a.averageHR ? Math.round(a.averageHR) : null, fcMax: a.maxHR ? Math.round(a.maxHR) : null,
    efecto: a.aerobicTrainingEffect ? Math.round(a.aerobicTrainingEffect * 10) / 10 : null,
  };
  if (tipo === 'fuerza') {
    act.reps = a.totalReps || 0;
    act.series = a.activeSets || a.totalSets || 0;
  }
  return { id: 'g' + act.id, dia, actividad: act };
}

// Descarga el .FIT de una sesión de fuerza o movilidad. Si se hizo con un entreno de Tangerine, devuelve la sesión ya apuntada.
function leerSesionGarmin_(token, g) {
  if (g.dia < PLAN.INICIO) return null;
  let fit;
  try {
    const zip = pedirGarmin_(token, '/download-service/files/activity/' + g.actividad.id).getBlob();
    const f = Utilities.unzip(zip.setContentType('application/zip')).filter(x => /\.fit$/i.test(x.getName()))[0];
    if (!f) return null;
    fit = FIT.leer(f.getBytes());
  } catch (err) {
    return null;
  }
  g.actividad.entreno = fit.entreno || '';
  const plan = ENTRENOS_GARMIN_[fit.entreno || g.actividad.nombre];
  if (!plan) return null;
  const E = PLAN.estado(registro_());
  const base = { id: 's' + g.actividad.id, dia: g.dia, para: PLAN.paraTipo(E, g.dia, plan), plan, garmin: g.actividad.id, reloj: true, minutos: g.actividad.min };
  const activas = fit.series.filter(x => x.activa);
  if (plan === 'test') {
    const prueba = { dominadas: 0, fondos: 0, flexiones: 0 };
    activas.forEach(x => { const id = PRUEBA_GARMIN_[x.cat + ':' + x.nombre]; if (id) prueba[id] = Math.max(prueba[id], x.reps || 0); });
    return Object.assign(base, { prueba });
  }
  if (plan === 'sab') return base;
  const series = {};
  activas.forEach(x => {
    const id = EJERCICIOS_GARMIN_[x.cat + ':' + x.nombre];
    if (!id) return;
    (series[id] = series[id] || []).push(PLAN.EJ[id].u === 's' ? (x.seg || 0) : (x.reps || 0));
  });
  // Escalón: si la app decía «hoy subes», las series ya son del escalón nuevo.
  const sube = PLAN.SESIONES[plan].filter(([, id, n]) => series[id] && PLAN.sugerir(PLAN.EJ[id], E.paso[id], E.ult[id], n).sube).map(([, id]) => id);
  return Object.assign(base, { series, sube });
}

// <fit>
// Lector mínimo de archivos .FIT de actividad: el deporte, el nombre del entreno y cada serie de fuerza.
// Solo usa JavaScript normal (funciona en Apps Script). Se copia en Codigo.gs con `node sincronizar.js`.
const FIT = (() => {
  const SESION = 18, ENTRENO = 26, SERIE = 225;
  function leer(bytes) {
    const b = bytes.map ? bytes.map(x => x & 0xFF) : Array.from(bytes, x => x & 0xFF);
    const cab = b[0];
    if (String.fromCharCode(b[8], b[9], b[10], b[11]) !== '.FIT') throw new Error('No es un archivo FIT');
    const fin = cab + (b[4] | b[5] << 8 | b[6] << 16 | b[7] << 24);
    const defs = {}, out = { deporte: null, subdeporte: null, entreno: '', series: [] };
    let i = cab;
    const num = (p, n, be) => { let v = 0; for (let k = 0; k < n; k++) v += b[p + (be ? n - 1 - k : k)] * Math.pow(256, k); return v; };
    while (i < fin) {
      const h = b[i++];
      if ((h & 0x80) === 0 && (h & 0x40)) {
        const be = b[i + 1] === 1, global = num(i + 2, 2, be), n = b[i + 4];
        i += 5;
        const campos = [];
        for (let k = 0; k < n; k++, i += 3) campos.push({ id: b[i], tam: b[i + 1], tipo: b[i + 2] & 0x1F });
        let dev = 0;
        if (h & 0x20) { const nd = b[i++]; for (let k = 0; k < nd; k++, i += 3) dev += b[i + 1]; }
        defs[h & 0x0F] = { global, campos, dev, be };
        continue;
      }
      const d = defs[(h & 0x80) ? (h >> 5) & 0x03 : h & 0x0F];
      if (!d) throw new Error('FIT raro: dato sin definición');
      const v = {};
      for (const c of d.campos) {
        if (d.global === ENTRENO && c.id === 8) {
          let s = '';
          for (let k = 0; k < c.tam && b[i + k]; k++) s += '%' + ('0' + b[i + k].toString(16)).slice(-2);
          try { v[c.id] = decodeURIComponent(s); } catch (e) { v[c.id] = ''; }
        } else if (c.tipo === 0x04 || c.tipo === 0x0B) { // uint16 (también listas)
          const l = [];
          for (let k = 0; k < c.tam; k += 2) l.push(num(i + k, 2, d.be));
          v[c.id] = l.length === 1 ? l[0] : l;
        } else if (c.tam === 4) v[c.id] = num(i, 4, d.be);
        else if (c.tam === 1) v[c.id] = b[i];
        i += c.tam;
      }
      i += d.dev;
      if (d.global === SESION) { out.deporte = v[5]; out.subdeporte = v[6]; }
      if (d.global === ENTRENO && v[8]) out.entreno = v[8];
      if (d.global === SERIE) {
        const pri = x => (Array.isArray(x) ? x[0] : x);
        const ok = x => (x == null || x === 0xFFFF || x === 0xFF || x === 0xFFFFFFFF ? null : x);
        out.series.push({
          activa: v[5] === 1,
          reps: ok(v[3]),
          seg: ok(v[0]) == null ? null : Math.round(v[0] / 1000),
          cat: ok(pri(v[7])),
          nombre: ok(pri(v[8])),
          paso: ok(v[11]),
        });
      }
    }
    return out;
  }
  return { leer };
})();
// </fit>

// Deja en el calendario exactamente lo que toca de hoy a DIAS_CALENDARIO días: crea, cambia o borra.
function sincronizar_() {
  const props = PropertiesService.getScriptProperties();
  const cal = CalendarApp.getCalendarById(props.getProperty('CAL') || '') || CalendarApp.getDefaultCalendar();
  const hoy = diaDe_(new Date());
  const desde = Utilities.parseDate(hoy + ' 00:00', TZ, 'yyyy-MM-dd HH:mm');
  const hasta = Utilities.parseDate(PLAN.diaMas(hoy, DIAS_CALENDARIO) + ' 00:00', TZ, 'yyyy-MM-dd HH:mm');
  const existentes = {};
  cal.getEvents(desde, hasta).forEach(ev => {
    const tag = ev.getTag('tangerine');
    if (tag) existentes[tag] ? ev.deleteEvent() : (existentes[tag] = ev);
  });
  const quiero = {};
  if (props.getProperty('CALENDARIO') !== 'no') {
    const E = PLAN.estado(registro_());
    for (let i = 0; i < DIAS_CALENDARIO; i++) {
      const x = PLAN.dia(E, PLAN.diaMas(hoy, i));
      if (!x.tipo || x.pausa || x.saltado) continue;
      quiero[x.dia] = {
        titulo: (x.hecho ? '✓ ' : '') + PLAN.titulo(x.tipo) + ' · Tangerine',
        inicio: Utilities.parseDate(x.dia + ' ' + ('0' + x.hora).slice(-5), TZ, 'yyyy-MM-dd HH:mm'),
        min: duracion_(E, x),
        desc: PLAN.detalle(E, x),
      };
    }
  }
  Object.keys(existentes).forEach(dia => { if (!quiero[dia]) existentes[dia].deleteEvent(); });
  Object.keys(quiero).forEach(dia => {
    const q = quiero[dia], fin = new Date(q.inicio.getTime() + q.min * 60000);
    let ev = existentes[dia];
    if (!ev) {
      ev = cal.createEvent(q.titulo, q.inicio, fin, { description: q.desc });
      ev.setTag('tangerine', dia);
      ev.setColor(COLOR_CAL);
      ev.removeAllReminders();
      ev.addPopupReminder(AVISO_MIN);
      return;
    }
    if (ev.getTitle() !== q.titulo) ev.setTitle(q.titulo);
    if (ev.getStartTime().getTime() !== q.inicio.getTime() || ev.getEndTime().getTime() !== fin.getTime()) ev.setTime(q.inicio, fin);
    if (ev.getDescription() !== q.desc) ev.setDescription(q.desc);
  });
}

function duracion_(E, x) {

function textoHoy_() {
  const E = PLAN.estado(registro_());
  const x = PLAN.dia(E, diaDe_(new Date()));
  if (x.pausa) return 'Hoy no entrenas: ' + x.pausa + '.';

    const base = Math.min(45, 30 + 5 * Math.floor(j / 2)), larga = Math.min(60, 35 + 5 * j);
    const k = descarga ? 0.67 : 1;
    return { fase: 2, sem, descarga, min: Math.round(base * k / 5) * 5, larga: Math.round(larga * k / 5) * 5,
      texto: `${Math.round(base * k / 5) * 5}′ fáciles · domingo ${Math.round(larga * k / 5) * 5}′${descarga ? ' · semana suave' : ''}` };
  }
  // Tramos del cronómetro: { n, s, k: 'run' | 'walk' }
  function tramosCorrer(sem, domingo) {
    const w = correrDe(sem), f = [{ n: 'Anda para calentar', s: 300, k: 'walk' }];
    if (w.fase === 1) {
      for (let i = 0; i < w.r; i++) {
        f.push({ n: 'Corre', s: w.c, k: 'run' });
        if (w.a && i < w.r - 1) f.push({ n: 'Anda', s: w.a, k: 'walk' });
      }
    } else f.push({ n: 'Corre suave', s: (domingo ? w.larga : w.min) * 60, k: 'run' });
    f.push({ n: 'Anda para soltar', s: 300, k: 'walk' });
    return f;
  }
  const tramosMov = id => MOV[id].l.map(([d, n, s]) => ({ d, n, s, k: d ? 'mov' : 'walk' }));
  const minutos = tramos => Math.round(tramos.reduce((a, x) => a + x.s, 0) / 60);

  // ---------- Calistenia ----------
  function sugerir(e, paso, ult, series) {
    const [lo, hi] = e.r, st = e.u === 's' ? 5 : 1;
    if (!ult || !ult.length) return { sube: false, obj: Array(series).fill(lo), nuevo: true };
    const u = Array.from({ length: series }, (_, i) => ult[i] != null ? ult[i] : ult[ult.length - 1]);
    if (u.every(v => v >= hi) && paso < e.esc.length - 1) return { sube: true, obj: u.map(() => lo) };
    return { sube: false, obj: u.map(v => Math.min(hi, v + st)) };
  }

  // ---------- Estado a partir del registro ----------
  // Tipos del registro: sesion, rodilla, mover, saltar, pausa, quitarPausa, deshacer.
  function estado(log) {
    const anulados = new Set(log.filter(e => e.tipo === 'deshacer').map(e => e.ref));
    const L = log.filter(e => e.tipo !== 'deshacer' && !anulados.has(e.id)).slice().sort((a, b) => (a.cuando < b.cuando ? -1 : a.cuando > b.cuando ? 1 : 0));
    const E = {
      paso: {}, ult: {}, hechos: {}, saltados: {}, movidos: [], pausas: PAUSAS.map(p => ({ ...p, fija: true })),
      correr: { sem: 0, hechas: 0, duros: 0 }, rodilla: [], pruebas: [{ ...PRUEBA_INICIAL, inicial: true }], subidas: [], muscleup: null,
      garmin: [], garminDia: {},
    };
    Object.keys(EJ).forEach(id => { E.paso[id] = EJ[id].paso; });
    const quitadas = new Set(), ignorados = new Set();
    for (const e of L) {
      if (e.tipo === 'sesion') {
        const para = e.para || e.dia;
        E.hechos[para] = e;
        if (e.plan === 'A' || e.plan === 'B') {
          // Al subir de escalón, las series de esa sesión ya son del escalón nuevo.
          (e.sube || []).forEach(id => {
            if (!EJ[id] || E.paso[id] >= EJ[id].esc.length - 1) return;
            E.paso[id]++;
            E.ult[id] = [];
            E.subidas.push({ dia: e.dia, id, a: EJ[id].esc[E.paso[id]] });
          });
          Object.keys(e.series || {}).forEach(id => {
            const v = (e.series[id] || []).filter(x => x != null);
            if (v.length && EJ[id]) E.ult[id] = v;
          });
        }
        if (e.plan === 'test') {
          E.pruebas.push({ dia: e.dia, dominadas: +e.prueba.dominadas || 0, fondos: +e.prueba.fondos || 0, flexiones: +e.prueba.flexiones || 0 });
          if (e.prueba.muscleup) E.muscleup = e.dia;
        }
        if (e.plan === 'run') {
          const c = E.correr, rod = Number(e.rodilla);
          E.rodilla.push({ dia: e.dia, v: rod, k: 'run' });
          if (rod > 3) { c.hechas = 0; c.duros = 0; }
          else {
            c.duros = e.sens === 'duro' ? c.duros + 1 : 0;
            if (c.duros >= 2) { c.hechas = 0; c.duros = 0; }
            else if (++c.hechas >= 3) { c.sem++; c.hechas = 0; c.duros = 0; }
          }
        }
      } else if (e.tipo === 'rodilla') {
        E.rodilla.push({ dia: e.dia, v: e.valor === 'peor' ? 'peor' : 'igual', k: 'manana' });
        if (e.valor === 'peor') E.correr.hechas = Math.max(0, E.correr.hechas - 1);
      } else if (e.tipo === 'mover') E.movidos.push({ de: e.de, a: e.a });
      else if (e.tipo === 'saltar') E.saltados[e.dia] = true;
      else if (e.tipo === 'pausa') E.pausas.push({ id: e.id, desde: e.desde, hasta: e.hasta, motivo: e.motivo || 'Pausa' });
      else if (e.tipo === 'quitarPausa') quitadas.add(e.ref);
      else if (e.tipo === 'garmin' && e.actividad) { E.garmin.push(e); (E.garminDia[e.dia] = E.garminDia[e.dia] || []).push(e.actividad); }
      else if (e.tipo === 'ignorar') ignorados.add(String(e.ref));
    }
    // Una actividad del reloj ya está apuntada si una sesión la lleva, si ese día ya hay una sesión de su clase o si se ignoró.
    const conReloj = new Set(L.filter(e => e.tipo === 'sesion' && e.garmin).map(e => String(e.garmin)));
    const claseDe = { run: 'run', fuerza: 'cal', flex: 'mov' };
    const diasClase = new Set(L.filter(e => e.tipo === 'sesion').map(e => e.dia + ':' + clase(e.plan)));
    E.garmin.forEach(g => {
      g.enlazado = conReloj.has(String(g.actividad.id)) || ignorados.has(g.id) || diasClase.has(g.dia + ':' + claseDe[g.actividad.tipo]) || !claseDe[g.actividad.tipo];
    });
    E.pausas = E.pausas.filter(p => !quitadas.has(p.id)).sort((a, b) => (a.desde < b.desde ? -1 : 1));
    // Cambios de día: intercambian lo que había en los dos días, en orden.
    E.asignado = {};
    E.movidos.forEach(m => { const x = tipoDia(E, m.de), y = tipoDia(E, m.a); E.asignado[m.a] = x; E.asignado[m.de] = y; });
    return E;
  }

  function pausaDe(E, dia) { return E.pausas.find(p => dia >= p.desde && dia <= p.hasta) || null; }
  function tipoBase(dia) {
    if (dia < INICIO) return null;
    const d = semanaDia(dia), b = SEMANA[d];
    if (b !== 'cal') return b;
    if (dia < INICIO_CAL) return null;
    const w = Math.floor(entre(INICIO_CAL, dia) / 7);
    if (d === 4 && (w + 1) % 4 === 0) return 'test';
    const orden = w % 2 === 0 ? ['A', 'B', 'A'] : ['B', 'A', 'B'];
    return orden[[0, 2, 4].indexOf(d)];
  }
  function tipoDia(E, dia) { return Object.prototype.hasOwnProperty.call(E.asignado, dia) ? E.asignado[dia] : tipoBase(dia); }

  // Qué toca un día: { dia, tipo, hora, hecho, saltado, pausa, movido }
  function dia(E, d) {
    const p = pausaDe(E, d);
    const tipo = p ? null : tipoDia(E, d);
    return {
      dia: d, tipo, hora: HORA[semanaDia(d)], pausa: p ? p.motivo : null, hecho: E.hechos[d] || null,
      saltado: !!E.saltados[d], movido: Object.prototype.hasOwnProperty.call(E.asignado, d),
    };
  }
  const semana = (E, lun) => Array.from({ length: 7 }, (_, i) => dia(E, diaMas(lun, i)));
  const esFuerte = tipo => ['A', 'B', 'test', 'run'].includes(tipo);

  // Sesiones de esta semana que pasaron sin hacer y sin saltar, con un hueco propuesto.
  function pendientes(E, hoy) {
    const out = [];
    for (let d = lunes(hoy); d < hoy; d = diaMas(d, 1)) {
      const x = dia(E, d);
      if (!esFuerte(x.tipo) || x.hecho || x.saltado) continue;
      let hueco = null;
      for (let h = hoy; h <= diaMas(lunes(hoy), 6); h = diaMas(h, 1)) {
        const y = dia(E, h);
        if ((y.tipo === 'sab' || y.tipo == null) && !y.pausa && !y.hecho) { hueco = h; break; }
      }
      out.push({ ...x, hueco });
    }
    return out;
  }

  function titulo(tipo) {
    return { A: 'Calistenia A', B: 'Calistenia B', test: 'Prueba', run: 'Correr', sab: 'Movilidad' }[tipo] || 'Descanso';
  }
  function clase(tipo) { return { A: 'cal', B: 'cal', test: 'cal', run: 'run', sab: 'mov' }[tipo] || 'off'; }
  function detalle(E, x) {
    if (x.pausa) return x.pausa;
    if (x.tipo === 'A' || x.tipo === 'B') return `${SESIONES[x.tipo].length} ejercicios en parejas · 45–50 min`;
    if (x.tipo === 'test') return 'Dominadas, fondos y flexiones al máximo · 20 min';
    if (x.tipo === 'run') return `Semana ${E.correr.sem + 1}${E.correr.sem < CORRER.length ? ' de 9' : ''} · ${correrDe(E.correr.sem).texto}`;
    if (x.tipo === 'sab') return `${minutos(tramosMov('sab'))} min · cadera, espalda y hombros`;
    return 'Descanso';
  }

  // Progreso hacia los hitos del muscle-up con las pruebas: ritmo por semana y fecha estimada.
  function hitos(E, hoy) {
    const P = E.pruebas, ult = P[P.length - 1], pri = P[0];
    const semanas = Math.max(1, entre(pri.dia, ult.dia) / 7);
    const ritmo = (ult.dominadas - pri.dominadas) / semanas;
    return HITOS.map(h => {
      let hecho = false, valor = null;
      if (h.mide === 'dominadas') { hecho = P.some(p => p.dominadas >= h.meta); valor = ult.dominadas; }
      if (h.mide === 'ambos') hecho = P.some(p => p.dominadas >= 8 && p.fondos >= 10);
      if (h.mide === 'muscleup') hecho = !!E.muscleup;
      let cuando = h.aprox;
      if (!hecho && h.mide !== 'muscleup' && P.length > 1 && ritmo > 0) {
        const meta = h.mide === 'ambos' ? 8 : h.meta;
        const d = diaMas(ult.dia, Math.ceil((meta - ult.dominadas) / ritmo * 7));
        cuando = '≈ ' + MESES[Number(d.slice(5, 7)) - 1] + ' ' + d.slice(0, 4);
      }
      return { ...h, hecho, valor, cuando };
    });
  }

  // Resumen de la semana que empieza en lun.
  function resumen(E, lun) {
    const dias = semana(E, lun), fin = diaMas(lun, 6);
    const plan = dias.filter(x => x.tipo && !x.pausa), hechas = dias.filter(x => x.hecho);
    const reps = l => l.reduce((a, x) => {
      const h = x.hecho;
      if (!h || !h.series) return a;
      return a + Object.keys(h.series).filter(id => EJ[id] && EJ[id].u !== 's').reduce((b, id) => b + h.series[id].reduce((c, v) => c + (v || 0), 0), 0);
    }, 0);
    const runs = hechas.filter(x => x.hecho.plan === 'run');
    return {
      lun, fin, plan: plan.length, hechas: hechas.length, reps: reps(dias), repsAntes: reps(semana(E, diaMas(lun, -7))),
      subidas: E.subidas.filter(s => s.dia >= lun && s.dia <= fin),
      salidas: runs.length, corriendo: runs.reduce((a, x) => a + (x.hecho.corriendo || 0), 0),
      rodilla: E.rodilla.filter(r => r.dia >= lun && r.dia <= fin),
      siguiente: semana(E, diaMas(lun, 7)),
    };
  }

  // Qué día del plan cubre una sesión de tipo `tipo` hecha el día d: ese mismo si tocaba eso;
  // si no, el primero de esa semana que tocara y siga pendiente; si no hay, el propio día.
  function paraTipo(E, d, tipo) {
    const x = dia(E, d);
    if (x.tipo === tipo && !x.hecho) return d;
    for (let i = 0; i < 7; i++) {
      const y = dia(E, diaMas(lunes(d), i));
      if (y.tipo === tipo && !y.hecho && !y.saltado && !y.pausa) return y.dia;
    }
    return d;
  }
  const paraCarrera = (E, d) => paraTipo(E, d, 'run');
  // Nombre del entreno del reloj para cada sesión (los archivos de garmin/entrenos).
  function entrenoReloj(E, tipo) {
    if (tipo === 'A' || tipo === 'B') return 'Calistenia ' + tipo;
    if (tipo === 'test') return 'Prueba máximos';
    if (tipo === 'sab') return 'Movilidad sábado';
    if (tipo === 'run') return E.correr.sem < CORRER.length ? 'Correr S' + (E.correr.sem + 1) : null;
    return null;
  }
  // Lo que se pone en Correr/Caminar del reloj esa semana (null si es carrera seguida).
  function reloj(sem) {
    const w = correrDe(sem);
    return w.fase === 1 && w.a ? { correr: fm(w.c), caminar: fm(w.a) } : null;
  }
  const ritmo = s => s ? Math.floor(s / 60) + ':' + String(s % 60).padStart(2, '0') + '/km' : '';

  function faseDe(d) { let f = FASES[0]; FASES.forEach(x => { if (d >= x.desde) f = x; }); return f; }

  return {
    INICIO, INICIO_CAL, CARRERA, SEMANA, HORA, DIAS, MESES, EJ, SESIONES, PRUEBA, CORRER, FASES, MOV,
    diaMas, entre, semanaDia, lunes, corta, fm, correrDe, tramosCorrer, tramosMov, minutos, sugerir,
    estado, dia, semana, pendientes, titulo, clase, detalle, hitos, resumen, faseDe, pausaDe, esFuerte, paraTipo, paraCarrera, entrenoReloj, reloj, ritmo,
  };
})();
// </plan>

// ---------- Hoja ----------

let LIBRO_ = null;

function libro_() {
  if (!LIBRO_) {
    const id = PropertiesService.getScriptProperties().getProperty('HOJA');
    if (!id) throw new Error('Falta ejecutar configurar()');
    LIBRO_ = SpreadsheetApp.openById(id);
  }
  return LIBRO_;
}

function hoja_(nombre) {
  const h = libro_().getSheetByName(nombre);
  if (!h) throw new Error('Falta la pestaña ' + nombre + ': ejecuta configurar()');
  return h;
}

function cabecera_(h) {
  return h.getRange(1, 1, 1, h.getLastColumn()).getValues()[0].map(String);
}

// Filas como objetos { columna: valor, _fila }. Se saltan las que tienen vacía la primera columna.
function leer_(nombre) {
  const v = hoja_(nombre).getDataRange().getValues();
  const cab = v[0].map(String);
  const out = [];
  for (let i = 1; i < v.length; i++) {
    if (v[i][0] === '' || v[i][0] == null) continue;
    const o = { _fila: i + 1 };
    cab.forEach((c, j) => { o[c] = v[i][j]; });
    out.push(o);
  }
  return out;
}

function anadir_(nombre, obj) {
  const h = hoja_(nombre);
  h.appendRow(cabecera_(h).map(c => celda_(obj[c])));
}

// ---------- Utilidades ----------

function claveValida_(k) {
  return !!k && k === PropertiesService.getScriptProperties().getProperty('CLAVE');
}

function comprobar_(k) {
  if (!claveValida_(k)) throw new Error('Clave incorrecta');
}

function texto_(s) {
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.TEXT);
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// Texto que entra en la hoja: sin espacios sobrantes y sin convertirse en fórmula ("=algo").
function limpiar_(s) {
  s = String(s == null ? '' : s).trim();
  return /^[=+\-@]/.test(s) ? "'" + s : s;
}

function celda_(v) {
  if (v == null) return '';
  return typeof v === 'string' ? limpiar_(v) : v;
}

function diaDe_(fecha) {
  return Utilities.formatDate(fecha, TZ, 'yyyy-MM-dd');
}
